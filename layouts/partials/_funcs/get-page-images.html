{{- /* Override of Hugo's embedded _funcs/get-page-images partial, which feeds the og:image / twitter:image tags emitted by _internal/opengraph.html and _internal/twitter_cards.html (called from Blowfish's head.html). Only change from upstream: a page's featured image is served as a 1200x630 crop instead of the full-size original. Pages without a featured image fall back to site.Params.images, as upstream does. */ -}} {{- $imgs := slice }} {{- with .Resources.ByType "image" }} {{- $featured := .GetMatch "*feature*" | default (.GetMatch "{*cover*,*thumbnail*}") }} {{- with $featured }} {{- $img := . }} {{- if ne .MediaType.SubType "svg" }}{{ $img = .Fill "1200x630 q75 Center" }}{{ end }} {{- $imgs = $imgs | append (dict "Image" $img "RelPermalink" $img.RelPermalink "Permalink" $img.Permalink) }} {{- end }} {{- end }} {{- if not $imgs }} {{- with site.Params.images }} {{- with index . 0 }} {{- $imgs = $imgs | append (dict "RelPermalink" (relURL .) "Permalink" (absURL .)) }} {{- end }} {{- end }} {{- end }} {{- return $imgs }}
